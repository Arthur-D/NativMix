```python
def _device_loop(self, inport, outport, target_device: str) -> None:
    """Poll a physical MIDI input (and optional output) until reconnect is needed."""
    while self._running and not self._panic_flag:
        if self._input_mode == "usb" or self._device_name != target_device:
            break
        self._process_pending_sync(outport)
        self._process_pending_mute_feedback(outport)
        self._flush_due_cc_volumes()
        drained = 0
        while True:
            msg = inport.receive(block=False)
            if msg is None:
                break
            drained += 1
            if msg.type == "control_change":
                self._handle_cc(int(msg.channel), msg.control, msg.value)
        if drained == 0:
            time.sleep(0.01)
```

Keep virtual-port loop sleep at `0.01` (already). Do **not** change `CcVolumeThrottler(0.02)` in this task.

```python
def _crossfader_side_names(self) -> tuple[list[str], list[str]]:
    a: list[str] = []
    b: list[str] = []
    for w in self._channels:
        side = (self._config.get_cross_side(w.channel_index) or "none").lower()
        name = self._channel_display_name(w.channel_index)
        if side == "a":
            a.append(name)
        elif side == "b":
            b.append(name)
    return a, b
```

```python
def _hit_scroll_chrome(self, event) -> bool:
    hit = self.childAt(event.position().toPoint())
    w = hit
    while w is not None and w is not self:
        from PyQt6.QtWidgets import QScrollBar
        if isinstance(w, QScrollBar):
            return True
        w = w.parentWidget()
    return False

def mousePressEvent(self, event) -> None:
    if event.button() == Qt.MouseButton.LeftButton:
        if self._hit_strip_drag_handle(event) or self._hit_scroll_chrome(event):
            super().mousePressEvent(event)
            return
        if self.windowHandle():
            self.windowHandle().startSystemMove()
    super().mousePressEvent(event)
```
